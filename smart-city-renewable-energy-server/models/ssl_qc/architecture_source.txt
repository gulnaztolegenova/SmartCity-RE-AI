def sha256(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for chunk in iter(lambda: stream.read(2**20), b''): h.update(chunk)
    return h.hexdigest()

# Notebook cell: fit-preprocessing
def fit_signal_preprocessor(frame: pd.DataFrame, signal_cols: list[str],
                            q_low: float = 0.01, q_high: float = 0.99) -> dict:
    """Fit every learned transformation on permitted fit rows only."""
    if not 0 <= q_low < q_high <= 1: raise ValueError('Invalid quantiles.')
    params={'cols':list(signal_cols),'q_low':q_low,'q_high':q_high,
            'winsorize':bool(CFG['preprocessing']['winsorize']),
            'sign_log1p':bool(CFG['preprocessing']['sign_log1p']),
            'fit_obs_ids_hash':hashlib.sha256(frame.obs_id.to_numpy(np.int64).tobytes()).hexdigest(),
            'fit_start':str(frame.timestamp.min()),'fit_end':str(frame.timestamp.max()),'features':{}}
    for col in signal_cols:
        s=frame[col].to_numpy(float)
        if np.isinf(s).any(): raise ValueError('Inf is not a missing-value convention.')
        observed=s[np.isfinite(s)]
        if len(observed)<2: raise ValueError(f'No usable training values: {col}')
        lo,hi=np.quantile(observed,[q_low,q_high]); median=float(np.median(observed))
        transformed=np.clip(observed,lo,hi) if params['winsorize'] else observed.copy()
        if params['sign_log1p']: transformed=np.sign(transformed)*np.log1p(np.abs(transformed))
        low=float(transformed.min()); high=float(transformed.max())
        # Same zero-span convention as original Fragment 6: use 1, not epsilon everywhere.
        denominator=(high-low) if high>low else 1.0
        params['features'][col]={'clip_lo':float(lo),'clip_hi':float(hi),'median':median,
                               'min':low,'max':high,'denom':denominator,'constant':bool(high==low)}
    return params

def transform_frame(frame: pd.DataFrame, params: dict) -> pd.DataFrame:
    """Keep row identity; never learn parameters or invent dates during transform."""
    out=pd.DataFrame(index=frame.index)
    for col in params['cols']:
        p=params['features'][col]; a=frame[col].to_numpy(float)
        if np.isinf(a).any(): raise ValueError('Infinite signal.')
        if params['winsorize']: a=np.clip(a,p['clip_lo'],p['clip_hi'])
        a=np.where(np.isnan(a),p['median'],a)
        if params['sign_log1p']: a=np.sign(a)*np.log1p(np.abs(a))
        out[col]=(a-p['min'])/p['denom']
    # Preserve original deterministic calendar encoding; not a learned MinMax fit.
    out['hour']=frame.timestamp.dt.hour/23.0
    out['month']=(frame.timestamp.dt.month-1)/11.0
    out['doy']=(frame.timestamp.dt.dayofyear-1)/365.0
    out=out[FEATURE_COLS].astype(np.float32)
    if not np.isfinite(out.to_numpy()).all(): raise ValueError('Non-finite preprocessed matrix.')
    return out

def inverse_signals(values: np.ndarray, params: dict) -> np.ndarray:
    """Inverse for ten signal channels only; winsorized extremes cannot be recovered."""
    values=np.asarray(values,dtype=float); out=np.empty_like(values)
    for j,col in enumerate(params['cols']):
        p=params['features'][col]; z=values[...,j]*p['denom']+p['min']
        out[...,j]=np.sign(z)*np.expm1(np.abs(z)) if params['sign_log1p'] else z
    if not np.isfinite(out).all(): raise FloatingPointError('Non-finite inverse reconstruction.')
    return out

# Notebook cell: prepare-windows
def make_windows_np(frame: pd.DataFrame, transformed: pd.DataFrame, length: int) -> dict:
    """Construct windows entirely within one partition; retain real row IDs and dates."""
    if len(frame)<length: raise ValueError('Too few rows for a complete window.')
    if not frame.index.equals(transformed.index): raise ValueError('Input row order mismatch.')
    ts=pd.DatetimeIndex(frame['timestamp'])
    if ts.hasnans or not ts.is_unique or not ts.is_monotonic_increasing:
        raise ValueError('Invalid timestamp alignment.')
    if not np.all(np.diff(ts.as_unit("ns").asi8)==pd.Timedelta(hours=1).value):
        raise ValueError('Windowing would cross a time gap.')
    def slide(a):
        return np.moveaxis(np.lib.stride_tricks.sliding_window_view(a,length,axis=0),-1,1).copy()
    values=transformed[FEATURE_COLS].to_numpy(np.float32)
    original=frame[FEATURE_COLS].to_numpy(np.float32)
    if np.isinf(original).any(): raise ValueError('Inf in original observations.')
    ids=frame.obs_id.to_numpy(np.int64)
    row_grid=np.lib.stride_tricks.sliding_window_view(ids,length).copy()
    pack={'W':slide(values),'observed':slide(np.isfinite(original)),
          'raw':slide(original),'row_ids':row_grid,'window_id':row_grid[:,0],
          'ts_start':np.asarray([t.isoformat() for t in ts[:len(ts)-length+1]],dtype='U40'),
          'ts_end':np.asarray([t.isoformat() for t in ts[length-1:]],dtype='U40')}
    assert pack['W'].shape==(len(frame)-length+1,length,len(FEATURE_COLS))
    assert np.all(pack['row_ids'][:,-1]-pack['row_ids'][:,0]==length-1)
    return pack

def exact_masks(pack: dict, rate: float, seed: int, repeat: int=0,
                pattern: str='random') -> np.ndarray:
    """Model-independent masks keyed by real window ID; stable under batching/subsetting.
    Natural missing entries never become ground-truth reconstruction labels.
    """
    if not 0<rate<1: raise ValueError('mask_rate must be strictly between zero and one.')
    observed=pack['observed']; n,length,d=observed.shape
    masks=np.zeros_like(observed,dtype=bool)
    for i,wid in enumerate(pack['window_id']):
        rng=np.random.default_rng(np.random.SeedSequence([int(seed),int(repeat),int(wid)]))
        valid=np.flatnonzero(observed[i].ravel())
        if len(valid)<2: raise ValueError('Not enough observed entries to mask.')
        k=min(max(1,int(round(rate*len(valid)))),len(valid)-1)
        if pattern=='random':
            chosen=rng.choice(valid,k,replace=False)
        elif pattern=='block':
            candidate=np.zeros((length,d),bool); span=min(length,max(1,int(np.ceil(rate*length))))
            for j in range(d):
                start=int(rng.integers(0,length-span+1)); candidate[start:start+span,j]=True
            selected=np.flatnonzero((candidate & observed[i]).ravel())
            if len(selected)>=k: chosen=rng.choice(selected,k,replace=False)
            else: chosen=np.r_[selected,rng.choice(np.setdiff1d(valid,selected),k-len(selected),replace=False)]
        elif pattern=='channel':
            # Realistic whole-channel failure: realized rate need not equal nominal rate.
            channels=rng.choice(d,max(1,min(d-1,int(round(rate*d)))),replace=False)
            if not np.any(channels<len(SIGNAL_COLS)):
                channels[0]=int(rng.integers(0,len(SIGNAL_COLS)))
            candidate=np.zeros((length,d),bool); candidate[:,channels]=True
            chosen=np.flatnonzero((candidate & observed[i]).ravel())
        elif pattern=='humidity_group':
            candidate=np.zeros((length,d),bool)
            humidity=[FEATURE_COLS.index(c) for c in ['humidity_solar','specific_humidity_solar','relative_humidity_solar']]
            steps=rng.choice(length,max(1,round(rate*length)),replace=False)
            candidate[np.ix_(steps,humidity)]=True
            for j in range(d):
                if j not in humidity:
                    candidate[rng.choice(length,max(1,round(rate*length)),replace=False),j]=True
            chosen=np.flatnonzero((candidate & observed[i]).ravel())
        else: raise ValueError(f'Unknown masking pattern: {pattern}')
        masks[i].ravel()[chosen]=True
    assert not (masks & ~observed).any()
    assert masks.sum()>0
    return masks

def masked_input(pack: dict, mask: np.ndarray, context: str='ALL') -> np.ndarray:
    x=pack['W'].copy()
    x[~pack['observed']]=0.0
    x[mask]=0.0
    if context in ('NO_TIME_INPUT','NO_TIME_RETRAIN'):
        x[:,:,len(SIGNAL_COLS):]=0.0
    elif context!='ALL': raise ValueError('Unknown context.')
    return x

def exact_mask_batch(x: torch.Tensor, observed: torch.Tensor, rate: float,
                     generator: torch.Generator) -> tuple[torch.Tensor,torch.Tensor]:
    """Training corruption with a dedicated CPU RNG, independent of validation RNG."""
    m=torch.zeros_like(observed,dtype=torch.bool,device='cpu')
    obs=observed.detach().cpu()
    for i in range(len(x)):
        valid=torch.nonzero(obs[i].reshape(-1),as_tuple=False).flatten()
        if len(valid)<2: raise ValueError('Insufficient observed values in training window.')
        k=min(max(1,round(rate*len(valid))),len(valid)-1)
        chosen=valid[torch.randperm(len(valid),generator=generator)[:k]]
        m[i].view(-1)[chosen]=True
    m=m.to(x.device); out=x.clone(); out[~observed]=0; out[m]=0
    return out,m

def metric_table(prediction: np.ndarray, pack: dict, mask: np.ndarray,
                 model: str, **ids) -> pd.DataFrame:
    """Pooled SSE/count, then square root. Never average batch RMSE."""
    prediction=np.asarray(prediction)
    if prediction.shape!=pack['W'].shape or not np.isfinite(prediction).all():
        raise ValueError('Invalid reconstruction.')
    mask=np.asarray(mask,bool) & pack['observed']
    err=prediction.astype(float)-pack['W'].astype(float)
    groups={'signals':list(range(len(SIGNAL_COLS))),
            'generation':[FEATURE_COLS.index(c) for c in TARGETS],
            'all_features':list(range(len(FEATURE_COLS)))}
    groups['nonredundant_signals']=[j for j,c in enumerate(SIGNAL_COLS) if c not in ['relative_humidity_solar','specific_humidity_solar']]
    groups.update({c:[FEATURE_COLS.index(c)] for c in TARGETS})
    rows=[]
    for group,indices in groups.items():
        selected=mask[:,:,indices]; e=err[:,:,indices][selected]
        n=int(len(e))
        rows.append(dict(ids,model=model,group=group,n_masked=n,
            MAE=float(np.abs(e).sum()/n) if n else np.nan,
            MSE=float(np.square(e).sum()/n) if n else np.nan,
            RMSE=float(np.sqrt(np.square(e).sum()/n)) if n else np.nan))
    return pd.DataFrame(rows)

def per_window_error(prediction: np.ndarray, pack: dict, mask: np.ndarray,
                     columns: Optional[list[int]]=None) -> tuple[np.ndarray,np.ndarray,np.ndarray]:
    columns=list(range(len(SIGNAL_COLS))) if columns is None else columns
    selected=(mask & pack['observed'])[:,:,columns]
    e=(prediction.astype(float)-pack['W'].astype(float))[:,:,columns]
    count=selected.sum((1,2)); squared=(e**2*selected).sum((1,2))
    absolute=(np.abs(e)*selected).sum((1,2))
    if np.any(count==0): raise ValueError('A window has no scored entries.')
    return squared,absolute,count

class DualAxisMaskedReconstructor(nn.Module):
    """Canonical DualAxis; sequential time attention followed by feature attention.
    Components are original Fragment 7; disabling an axis creates an ablation, not a new model.
    """
    def __init__(self,L:int,D:int,d_model:int,nhead:int,layers_time:int,layers_feat:int,
                 dropout:float,condition:str='FULL'):
        super().__init__();self.L=L;self.D=D;self.d_model=d_model;self.condition=condition
        self.val_proj=nn.Linear(1,d_model)
        self.time_pos=nn.Parameter(torch.zeros(L,d_model));self.feat_emb=nn.Parameter(torch.zeros(D,d_model))
        def encoder(depth):
            layer=nn.TransformerEncoderLayer(d_model,nhead,dim_feedforward=4*d_model,
               dropout=dropout,batch_first=True,activation='gelu',norm_first=True)
            return nn.TransformerEncoder(layer,num_layers=depth,enable_nested_tensor=False)
        self.temporal=encoder(layers_time) if condition!='FEATURE_ONLY' else nn.Identity()
        self.feature=encoder(layers_feat) if condition!='TIME_ONLY' else nn.Identity()
        self.out_proj=nn.Linear(d_model,1)
        nn.init.trunc_normal_(self.time_pos,std=.02);nn.init.trunc_normal_(self.feat_emb,std=.02)
    def tokens(self,x):
        b,length,d=x.shape
        if (length,d)!=(self.L,self.D): raise ValueError('DualAxis input shape does not match configuration.')
        tok=self.val_proj(x.unsqueeze(-1))+self.time_pos[None,:,None,:]+self.feat_emb[None,None,:,:]
        tok=self.temporal(tok.permute(0,2,1,3).reshape(b*d,length,self.d_model))
        tok=tok.reshape(b,d,length,self.d_model).permute(0,2,1,3).contiguous()
        tok=self.feature(tok.reshape(b*length,d,self.d_model))
        return tok.reshape(b,length,d,self.d_model)
    def forward(self,x): return self.out_proj(self.tokens(x)).squeeze(-1)
    def get_embedding(self,x): return self.tokens(x).mean(dim=(1,2))

@torch.no_grad()
def predict_neural(model:nn.Module,pack:dict,mask:np.ndarray,context:str='ALL',batch_size:Optional[int]=None) -> np.ndarray:
    batch_size=batch_size or CFG['evaluation']['batch_size'];model.eval()
    x=masked_input(pack,mask,context);out=[]
    for start in range(0,len(x),batch_size):
        batch=torch.from_numpy(x[start:start+batch_size]).to(device)
        y=model(batch)
        if y.shape!=batch.shape or not torch.isfinite(y).all(): raise FloatingPointError('Invalid reconstruction.')
        out.append(y.cpu().numpy())
    return np.concatenate(out)

@torch.no_grad()
def extract_embeddings(model:nn.Module,pack:dict,batch_size:Optional[int]=None,context:str='ALL') -> np.ndarray:
    batch_size=batch_size or CFG['qc']['batch_size'];model.eval();out=[]
    x=masked_input(pack,np.zeros_like(pack['observed']),context)
    for start in range(0,len(x),batch_size):
        z=model.get_embedding(torch.from_numpy(x[start:start+batch_size]).to(device))
        if z.ndim!=2 or not torch.isfinite(z).all(): raise ValueError('Invalid embedding.')
        out.append(z.cpu().numpy())
    return np.concatenate(out)

# Notebook cell: qc-definition
def robust_location_scale(reference: np.ndarray) -> dict:
    values=np.asarray(reference,float)
    if values.ndim!=1 or not np.isfinite(values).all(): raise ValueError('Invalid QC calibration values.')
    median=float(np.median(values));mad=float(np.median(np.abs(values-median)))
    scale=1.4826*mad;source='MAD'
    if scale<=1e-12:
        scale=float((np.quantile(values,.75)-np.quantile(values,.25))/1.349);source='IQR'
    if scale<=1e-12: scale=float(values.std());source='standard deviation'
    if scale<=1e-12: scale=1.;source='constant channel convention'
    return {'median':median,'scale':scale,'scale_source':source}

def apply_robust_z(values: np.ndarray, params: dict) -> np.ndarray:
    return (np.asarray(values)-params['median'])/params['scale']

def repeated_window_rmse(model: nn.Module,pack: dict,rate: float,seed: int,repeats: int,
                         context: str='ALL') -> np.ndarray:
    errors=[]
    for repeat in range(repeats):
        mask=exact_masks(pack,rate,seed,repeat)
        prediction=predict_neural(model,pack,mask,context,CFG['qc']['batch_size'])
        sse,_,count=per_window_error(prediction,pack,mask)
        errors.append(np.sqrt(sse/count))
    # This statistic intentionally matches window-level QC: mean of repeat RMSEs.
    return np.mean(errors,axis=0)

def fit_qc_system(model:nn.Module,fit:dict,cal:dict) -> dict:
    """Learn density on fit embeddings; reference-score scale and cutoff on validation, never test."""
    z_fit=extract_embeddings(model,fit);z_cal=extract_embeddings(model,cal)
    scaler=RobustScaler().fit(z_fit)
    zs=scaler.transform(z_fit)
    dim=min(CFG['qc']['embedding_pca_dim'],zs.shape[1],len(zs)-1)
    projection=PCA(n_components=dim,svd_solver='full').fit(zs)
    embedded=projection.transform(zs);cal_embed=projection.transform(scaler.transform(z_cal))
    gmm=GaussianMixture(n_components=CFG['qc']['gmm_components'],covariance_type='full',
        reg_covar=1e-4,n_init=3,max_iter=300,random_state=42).fit(embedded)
    if not gmm.converged_: raise RuntimeError('Density model did not converge.')
    km=KMeans(n_clusters=CFG['qc']['gmm_components'],n_init=20,random_state=42).fit(embedded)
    components={'rmse':repeated_window_rmse(model,cal,CFG['qc']['mask_rate'],42,CFG['qc']['mask_repeats']),
                'nll':-gmm.score_samples(cal_embed),
                'distance':np.min(km.transform(cal_embed),axis=1)}
    if CFG['qc']['mc_passes']>=2:
        components['uncertainty']=mean_masked_uncertainty(model,cal,CFG['qc']['mask_rate'],CFG['qc']['mc_passes'],42)
    norm={name:robust_location_scale(values) for name,values in components.items()}
    weights={name:float(CFG['qc']['weights'][name]) for name in components}
    total=sum(weights.values());weights={name:w/total for name,w in weights.items()}
    calibration_score=sum(weights[name]*apply_robust_z(values,norm[name]) for name,values in components.items())
    return {'scaler':scaler,'projection':projection,'gmm':gmm,'centroids':km,'normalization':norm,
            'weights':weights,'cutoff':float(np.quantile(calibration_score,CFG['qc']['calibration_quantile'])),
            'calibration_score':calibration_score,'calibration_components':components}

def qc_components(model,system,pack,rate,seed,repeats,context='ALL'):
    z=extract_embeddings(model,pack,context=context)
    z=system['projection'].transform(system['scaler'].transform(z))
    parts={'rmse':repeated_window_rmse(model,pack,rate,seed,repeats,context),
           'nll':-system['gmm'].score_samples(z),'distance':np.min(system['centroids'].transform(z),axis=1)}
    if 'uncertainty' in system['weights']:
        parts['uncertainty']=mean_masked_uncertainty(model,pack,rate,CFG['qc']['mc_passes'],seed,context)
    score=sum(system['weights'][key]*apply_robust_z(values,system['normalization'][key]) for key,values in parts.items())
    return score,parts
